```cpp
#include <iostream>

using namespace std;

const int N = 1e5+9;

int a[N],d[N];

int main()

{

  int m,n;

  while(cin >> n >> m)

  {

    for(int i = 1;i <= n; ++i) cin >>a[i];

    for(int i = 1;i <= n; ++i) d[i] = a[i] - a[i - 1];

  

    while(m--)

    {

      int l,r,v;

      cin >> l >> r >> v;

      d[l] += v,d[r+1] -= v;

    }

    for(int i = 1;i <= n;++i)

      {

        a[i] = a[i-1] + d[i];

      }

    for(int i = 1;i <= n;++i) cout << a[i] << " \n"[i == n];

  }

  
  
  

  return 0;

}
```
