#include<bits/stdc++.h>
using namespace std;
const int N = 1e5+9;
int a[N];

vector<int> L;

int main()
{
	int n;cin >> n;
	for(int i = 1;i <= n;++i) cin >> a[i];
	sort(a+1,a+1+n);
	for(int i = 1;i <= n;++i) L.push_back(a[i]);
	 
	L.erase(unique(L.begin(),L.end()),L.end());
	
	for(const &x : L) cout << x << " \n"[&x == &L.back()];
	
	int y;cin >> y;
	cout << lower_bound(L.begin(),L.end(),y) - L.begin();
	return 0;
}
