```cpp
   pq.push(x);

ll ans = 0;

 while(pq.size() > 1)

 {

   ll x = pq.top();pq.pop();

   ll y = pq.top();pq.pop();

   pq.push(x+y); 

   ans += x + y;

  

 }

}
```

```cpp
#include <bits/stdc++.h>

using namespace std;

const int N = 1e7+9;

int a[N];

int main()

{

  int w;cin >> w;

  int n;cin >> n;

  for(int i = 1;i <= n;++i)

  {

    cin >> a[i];

  }

  sort(a+1,a+1+n);

  int l = 1,r = n,ans = 0;

  while(l <= r)

  {

    if(l == r)

    {

      ans++;

      break;

    }

    if(a[l] + a[r] <= w) l++,r--,ans++;

    else r--,ans++;

  }

  cout << ans << '\n';

  return 0;

}
```
