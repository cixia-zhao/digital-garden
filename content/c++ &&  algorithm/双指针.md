对撞指针例题：

![[Pasted image 20260311164227.png]]

```cpp
#include <bits/stdc++.h>

using namespace std;

const int N = 1e6+9;

char s[N];

int main()

{

  ios::sync_with_stdio(0),cin.tie(0),cout.tie(0);

  cin >> s+1;

  

  int n = strlen(s+1);

  int l = 1,r = n;

  bool ans = true;

  while(l<r && ans)

  {

    if(s[l] != s[r]) ans = false;

    l++,r--;

  }

  cout << (ans ? 'Y' : 'N') <<'\n';

  return 0;

}
```

快慢指针（滑动窗口）例题：

![[Pasted image 20260311172137.png]]


```cpp
#include <bits/stdc++.h>

using namespace std;

const int N = 1e5+9;

int a[N]; 

int main()

{
  ios::sync_with_stdio(0),cin.tie(0),cout.tie(0);
  int n,s;cin >> n >> s;

  int l =1;

  int ans = n+1;

  for(int i = 1;i <= n;++i) cin >> a[i];

  int count = 0;

  for(int r = 1;r<=n;++r)

  {

    count += a[r];

    while(count >= s)

    {

      ans = min(ans,r-l+1);

      count -= a[l];

      l++;

    }

  }

  cout << (ans <= n ? ans : 0) <<'\n'; 

  return 0;

}
```

```cpp
#include <bits/stdc++.h>
using namespace std;
const int N = 1e4+9;
int a[N];

int main()

{

  ios::sync_with_stdio(0),cin.tie(0),cout.tie(0);

  int n,m,k;cin >> n >> m >> k;

  for(int i = 1;i <= n;++i) cin >> a[i]; 

  int l =1;

  int cnt = 0;

  int ans = 0;

  for(int r = 1;r<=n;++r)

  {

    if(a[r] >= m)

    {

      cnt++;

    }

    while(cnt >= k) 

    {

      ans += n - r + 1;

      if(a[l] >= m) 

      {

        cnt--;

      }

      l++;

    }

  }

  cout << ans <<'\n';

  return 0;

}
```
